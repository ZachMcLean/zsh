# Environment variables set everywhere
export EDITOR="nvim"
export VISUAL="nvim"

if [[ -n $DISPLAY ]]; then
  # Speedy Keys
  command -v xset >/dev/null && xset r rate 210 40
  export TERMINAL="ghostty"
  export BROWSER="firefox"
fi

# # XDG PATHS
export XDG_CONFIG_HOME=$HOME/.config
export XDG_CACHE_HOME=$HOME/.cache
export XDG_DATA_HOME=$HOME/.local/share

[[ -x /opt/homebrew/bin/brew ]] && eval "$(/opt/homebrew/bin/brew shellenv)"
typeset -U path PATH 
path=("$HOME/.local/bin" $path)
export PNPM_HOME="$XDG_DATA_HOME/pnpm" && path=("$PNPM_HOME" $path)

# ~/ Clean-up:
export LESSHISTFILE="-"
export ZDOTDIR="$XDG_CONFIG_HOME/zsh"

# Pfetch Settings
export PF_SEP=":"
export PF_INFO="ascii title os kernel wm editor shell pkgs memory"

[[ -d /Library/Frameworks/Python.framework/Versions/3.12/bin ]] && path=("/Library/Frameworks/Python.framework/Versions/3.12/bin" $path)
